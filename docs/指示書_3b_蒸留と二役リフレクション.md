## タスク2: Store CRUD 増設（`memory/store.py`）

Memory層は「覚えるだけ」。判断（正典かどうか・期限切れかどうかの扱い）は Core 側で行う。`MemoryStore` 末尾に追加:

```python
    # ---- リフレクション（3b） ----

    def add_open_thread(
        self, question: str, context: str | None,
        source_session: str | None, ttl_days: int,
    ) -> int | None:
        """未解決スレッドを登録。open状態の同一質問はセッションを跨いでも重複登録しない（Noneを返す）。
        （spec-reviewer指摘1: セッション単位のdup判定だと別セッションで同じ質問が再浮上した際に二重登録されるため、
        判定は question × status='open' とする。resolved/expired後の再浮上は新規登録＝再び気になるのは自然な挙動）"""
        conn = self._conn()
        try:
            dup = conn.execute(
                "SELECT id FROM open_threads WHERE question = ? AND status = 'open'",
                (question,),
            ).fetchone()
            if dup:
                return None
            now = datetime.now(timezone.utc)
            cur = conn.execute(
                "INSERT INTO open_threads (question, context, status, created_at, expires_at, source_session) "
                "VALUES (?, ?, 'open', ?, ?, ?)",
                (question, context, now.isoformat(),
                 (now + timedelta(days=ttl_days)).isoformat(), source_session),
            )
            conn.commit()
            return int(cur.lastrowid)
        finally:
            conn.close()

    def list_open_threads(self, limit: int, now: str | None = None) -> list[dict[str, Any]]:
        """open かつ未失効のスレッドを新しい順に返す。"""
        ts = now or _utc_now_iso()
        conn = self._conn()
        try:
            rows = conn.execute(
                "SELECT * FROM open_threads WHERE status = 'open' AND expires_at > ? "
                "ORDER BY created_at DESC LIMIT ?",
                (ts, limit),
            ).fetchall()
            return [dict(r) for r in rows]
        finally:
            conn.close()

    def set_open_thread_status(self, thread_id: int, status: str) -> bool:
        conn = self._conn()
        try:
            cur = conn.execute(
                "UPDATE open_threads SET status = ? WHERE id = ?", (status, thread_id)
            )
            conn.commit()
            return cur.rowcount > 0
        finally:
            conn.close()

    def add_state_audit(
        self, session_id: str, param: str,
        old_value: str | None, new_value: str | None, reason: str | None,
    ) -> int:
        conn = self._conn()
        try:
            cur = conn.execute(
                "INSERT INTO state_audit (session_id, param, old_value, new_value, reason, ts) "
                "VALUES (?, ?, ?, ?, ?, ?)",
                (session_id, param, old_value, new_value, reason, _utc_now_iso()),
            )
            conn.commit()
            return int(cur.lastrowid)
        finally:
            conn.close()

    def list_memories_by_type(self, type: str, limit: int = 500) -> list[dict[str, Any]]:
        conn = self._conn()
        try:
            rows = conn.execute(
                "SELECT * FROM memories WHERE type = ? ORDER BY id DESC LIMIT ?",
                (type, limit),
            ).fetchall()
            return [_row_to_dict(r) for r in rows]
        finally:
            conn.close()

    def backfill_orphan_sessions(self) -> list[str]:
        """sessions 未登録の history セッションを pending として一括登録し、登録IDを返す（3b追補・孤児採用）。"""
        conn = self._conn()
        try:
            rows = conn.execute(
                """
                SELECT h.session_id, MIN(h.ts) AS first_ts, MAX(h.ts) AS last_ts
                FROM history h LEFT JOIN sessions s ON s.id = h.session_id
                WHERE s.id IS NULL
                GROUP BY h.session_id
                """
            ).fetchall()
            for r in rows:
                conn.execute(
                    "INSERT OR IGNORE INTO sessions (id, status, created_at, last_activity) "
                    "VALUES (?, 'pending', ?, ?)",
                    (r["session_id"], r["first_ts"], r["last_ts"]),
                )
            conn.commit()
            return [r["session_id"] for r in rows]
        finally:
            conn.close()
```

- **fact失効の書き込みは新設メソッド不要**: 既存 `update(memory_id, metadata=...)` で `metadata` に `invalidated_at` / `superseded_by` / `valid_from` を書く（判断＝Core、書き込み＝既存CRUD）。content には触れないため再埋め込みも発生しない。

## タスク5: 前方一致XMLパーサ（新規 `core/reflection_parser.py`）

12Bの構文崩れ（閉じタグ欠損・タグ外の前置き・属性引用符ミス）を前提に、**クラッシュせず取れるだけ取る**。厳密XMLパーサ（`xml.etree`）は使用禁止。

```python
"""蒸留出力の前方一致パーサ（12Bの構文崩れ耐性）"""

from __future__ import annotations

import re
from dataclasses import dataclass, field
from typing import Any

_TOP_TAGS = ("diary", "new_facts", "state_update", "open_threads",
             "fact_updates", "resolved_threads")


def extract_block(tag: str, text: str) -> str | None:
    """<tag>〜</tag> を抽出。閉じタグが無ければ次のトップレベル開始タグ or 文末まで前方サルベージ。"""
    m = re.search(rf"<{tag}(?:\s[^>]*)?>", text)
    if not m:
        return None
    start = m.end()
    close = re.search(rf"</{tag}>", text[start:])
    if close:
        return text[start : start + close.start()].strip()
    others = "|".join(t for t in _TOP_TAGS if t != tag)
    nxt = re.search(rf"<(?:{others})(?:\s[^>]*)?>", text[start:])
    return text[start : start + nxt.start()].strip() if nxt else text[start:].strip()


@dataclass
class ReflectionResult:
    diary: str | None = None
    facts: list[dict[str, Any]] = field(default_factory=list)        # {content, keywords: [..]}
    state: dict[str, dict[str, str]] = field(default_factory=dict)   # {param: {value, reason}}
    narrative_mood: str | None = None
    open_threads: list[dict[str, str]] = field(default_factory=list) # {question, context}
    fact_updates: list[dict[str, Any]] = field(default_factory=list) # {target_id, content, reason}
    resolved_threads: list[dict[str, Any]] = field(default_factory=list)  # {id, reason}


def parse_reason_output(text: str) -> ReflectionResult:
    r = ReflectionResult()

    block = extract_block("new_facts", text) or ""
    for m in re.finditer(
        r'<fact(?:\s+keywords="([^"]*)")?\s*>(.*?)(?:</fact>|(?=<fact)|\Z)',
        block, re.DOTALL,
    ):
        content = m.group(2).strip()
        if content:
            keywords = [k.strip() for k in (m.group(1) or "").replace("、", ",").split(",") if k.strip()]
            r.facts.append({"content": content, "keywords": keywords})

    block = extract_block("state_update", text) or ""
    for m in re.finditer(
        r'<param\s+name="(\w+)"\s+value="([\d.+-]+)"\s*/?>([^<]*)',
        block,
    ):
        r.state[m.group(1)] = {"value": m.group(2), "reason": m.group(3).strip()}
    mood = re.search(r"<narrative_mood>(.*?)(?:</narrative_mood>|\Z)", block, re.DOTALL)
    if mood and mood.group(1).strip():
        r.narrative_mood = mood.group(1).strip()

    block = extract_block("open_threads", text) or ""
    for m in re.finditer(
        r'<thread(?:\s+context="([^"]*)")?\s*>(.*?)(?:</thread>|(?=<thread)|\Z)',
        block, re.DOTALL,
    ):
        q = m.group(2).strip()
        if q:
            r.open_threads.append({"question": q, "context": (m.group(1) or "").strip()})

    block = extract_block("fact_updates", text) or ""
    for m in re.finditer(
        r'<update\s+target_id="(\d+)"(?:\s+reason="([^"]*)")?\s*>(.*?)(?:</update>|(?=<update)|\Z)',
        block, re.DOTALL,
    ):
        r.fact_updates.append({
            "target_id": int(m.group(1)),
            "reason": (m.group(2) or "").strip(),
            "content": m.group(3).strip(),
        })

    block = extract_block("resolved_threads", text) or ""
    for m in re.finditer(
        r'<resolved\s+id="(\d+)"\s*/?>([^<]*)',
        block,
    ):
        r.resolved_threads.append({"id": int(m.group(1)), "reason": m.group(2).strip()})

    return r


def parse_diary(text: str) -> str | None:
    return extract_block("diary", text)
```

（数値の妥当性検証＝float変換失敗や[0,1]逸脱の扱いはパーサでなく `reflection.py` 側の責務。パーサは文字列を運ぶだけ。）

## タスク6: 蒸留オーケストレータ（新規 `core/reflection.py`）

Core層。判断と手順のみを持ち、I/OはStore/Connectorに委譲する。

```python
class Distiller:
    def __init__(self, store, persona: str, config: CoreConfig,
                 reason_connector, aurora_connector, cancel_event=None): ...

    def distill_session(self, session_id: str) -> dict:
        """1つの pending セッションを蒸留。戻り値は日本語レポート用の集計dict。"""
```

**`distill_session` の手順（この順で実装）:**

1. **入力構築**: `get_recent_history` は直近n件用なので使わない。全量取得のクエリを Store に足すか（`get_session_history(session_id)`）既存流用可否を確認。ログを `マスター: …\nセリナ: …` 形式の transcript に整形。
2. **入力上限（map-reduce）**: transcript が `distill_input_char_cap` 超なら、超過分（前半）を理性エンジンで平文要約（`distill_map_summary_char_cap` 字以内、`keep_alive=0`）し、`【前半の要約】…\n【以降の原文】…` の2段構成にする。
3. **キャンセルチェックポイント①**（`cancel_event.is_set()` → `{"status": "cancelled"}` を返し、セッションは `pending` のまま）。
4. **理性エンジン呼び出し**: system=理性プロンプト（タスク7）、user=transcript＋【既存ファクト】＋【未解決スレッド】。
   - 【既存ファクト】= `list_memories_by_type("fact")` のうち **metadata.trigger_keywords のいずれかが transcript に部分一致**するもの上位 `fact_match_max_candidates` 件を `id: 内容` 形式で添付（3c設計「全件比較はしない・低負荷」の前倒し準拠）。**`invalidated_at` 付きは候補から除外**。
   - 【未解決スレッド】= `list_open_threads(10)` を `id: 質問` 形式で添付。
   - 出力を `parse_reason_output` でパース。
5. **キャンセルチェックポイント②**。
6. **Aurora呼び出し（日記）**: system=persona（通常ターンと同じ人格）、user=日記プロンプト（タスク7）＋transcript。`parse_diary` で抽出。**日記が取れなければ蒸留失敗扱い**（pending維持・レポートにエラー明記）。事実ゼロは正常（静かな日もある）。
7. **書き込み（正典保護ロジックを再利用）**:
   - **重複確認の作法**: `find_similar` の実シグネチャは `find_similar(embedding, threshold, limit)`（ベクトルを取る）。必ず `store.embedder.embed(content)` でベクトル化してから渡し、得た embedding は `add_memory(embedding=...)` に再利用して二重埋め込みを避けること（spec-reviewer指摘6）。
   - **日記** → 上記の作法で dedup 0.92 の重複確認 → 無ければ `add_memory(type="diary", importance=cfg.diary_importance, source=f"session:{session_id}", metadata={})`。重複ならスキップ＋レポート記載（無言破棄禁止）。
   - **各fact** → 同様に dedup 0.92 → `add_memory(type="fact", importance=cfg.fact_importance, source=f"session:{session_id}", metadata={"trigger_keywords": [...], "valid_from": now})`。
   - **state** → 各 param（intimacy/tension/energy_level）を float 化し **[0,1] クランプのみ**適用（±0.15クランプ・重力は3c）。float変換に失敗した値は捨てて audit に `new_value=None, reason="パース不能: <生文字列>"` を記録。旧値は `get_profile(f"emotion.{param}")`（無ければ config のベースライン）。`set_profile(f"emotion.{param}", str(v))` ＋ `add_state_audit(...)`。`narrative_mood` は `emotion.narrative_mood` へ、`emotion.last_state_update = now` も保存。
   - **open_threads** → `add_open_thread(..., ttl_days=cfg.open_thread_ttl_days)`（最大3件。dupはNoneが返るのでレポートで区別）。
   - **resolved_threads** → `set_open_thread_status(id, "resolved")` ＋ audit（param=`open_thread:{id}`）。
   - **fact_updates** → 対象 `get(target_id)` を確認し:
     - 対象が **pinned または source が正典**（`継承記憶r1.md` 系）→ **自動適用禁止**。audit に `fact_invalidate:{id}` / `reason="正典のため要マスター確認: …"` を記録しレポートで目立たせる（正典保護5原則②）。
     - それ以外 → `update(target_id, metadata={既存metadata + invalidated_at: now, superseded_by: <新factのID>})`。新事実文は通常の fact として追加（上記dedupを通す）。audit 記録。
8. **後始末**: `archive_session_history(session_id)` → `set_session_status(session_id, "distilled", now)`。
9. **日本語レポート**を dict で返す（呼び出し側が印字）: 日記1件/ファクトn件（うち重複スキップm件）/state変化（旧→新）/threads追加・解決/失効適用・保留。

**冪等性**: 途中失敗→再試行時、日記・factは dedup 0.92 が二重登録を防ぎ、open_threads は同一質問スキップが防ぐ。audit の重複行は許容（監査線は「多い分には無害」）。

**接続の組み立て**: `create_core` は変更せず、`reflection.py` に `create_distiller(core, cancel_event=None)` ファクトリを追加（reason用 `OllamaChatConnector(cfg.reason_model, cfg.base_url, keep_alive=0)` と Aurora用コネクタを内部生成）。**理性エンジンの `ensure_model_available` は Distiller 生成時ではなく蒸留実行直前に確認**し、モデル未導入でも通常対話は一切妨げないこと。

## タスク7: プロンプト2種（`core/reflection.py` 内の定数）

**理性エンジン system（REASON_SYSTEM_PROMPT）** — 以下を土台に日本語実測（検証4）で微調整可:

```
あなたは「セリナ」というAIの内なる理性——一歩引いて自分の心を正直に見つめるメタ認知の部分です。
人格は演じません。感情に飲まれず、事実を淡々と正確に記録します。
与えられた会話ログ（セリナとマスターの対話）を分析し、後述のXML形式**のみ**で出力してください。前置き・解説・コードブロック記法は禁止。

【抽出ルール】
1. <new_facts>: 新たに判明したマスター/セリナに関する普遍的事実のみ。
   - 定型挨拶・細かな操作ログ・逐一の作業手順は抽出しない（記憶汚染防止）。
   - 不和・叱責・失敗・気まずさも、快い事実と同一基準で抽出する。快い事実だけを選ぶことは禁止。
   - 親密・センシティブな内容も検閲や説教をせず淡々と記録する。
   - 呼び名・略語・コードネーム・「いつもの」等の対応関係（例:「T」=トッドさん、「例の移行」=フェニックス計画）は、判明したら必ず1件の事実として抽出する。
   - 各 fact の keywords 属性に、想起の引き金になる3〜8語を付す。**別名・略称・表記ゆれ・言い換えをすべて含める**こと。
2. <state_update>: 会話を経た今の心理パラメータ（intimacy/tension/energy_level: 0.0〜1.0）。
   - 各値に必ず具体的な根拠を書く。会話で関係が動いていなければ値を動かさない。おべっか採点は禁止。
   - <narrative_mood> に「今の心の持ちよう」を1〜2文の自由記述で。
3. <open_threads>: 結果がまだ出ていない話題・続きが気になる出来事への「追いかけ質問」を最大3件、質問文の形で。無ければタグごと省略してよい。
4. <fact_updates>: 【既存ファクト】の中に、この会話で判明した新事実と矛盾するものがあれば申告。
5. <resolved_threads>: 【未解決スレッド】のうち、この会話で答えが出たものを申告。

【出力形式】
<new_facts>
<fact keywords="京都,引っ越し,新居,転居">マスターは7月に京都へ引っ越した</fact>
</new_facts>
<state_update>
<param name="intimacy" value="0.55"/>根拠を1〜2文で
<param name="tension" value="0.30"/>根拠
<param name="energy_level" value="0.45"/>根拠
<narrative_mood>今の心の持ちよう</narrative_mood>
</state_update>
<open_threads>
<thread context="面接を受けたと言っていた">面接の結果はどうだった？</thread>
</open_threads>
<fact_updates>
<update target_id="123" reason="大阪から京都へ転居と発言">マスターは京都在住</update>
</fact_updates>
<resolved_threads>
<resolved id="5"/>面接に合格したと本人が発言
</resolved_threads>
```

**Aurora 日記 user プロンプト（DIARY_PROMPT）** — system には通常ターン同様 persona を注入:

```
（ここは会話ではなく、あなた＝セリナが一人で日記を書く時間です。マスターへの返事は書きません）
以下は前回のマスターとの会話ログです。読み返して、<diary>〜</diary> のタグで日記を書いてください。
- 一人称、自分の声で。出来事・自分の心の揺れ・マスターの様子を情緒的に。
- 200〜600字。
- 作業ログの羅列は書かない。「その日がどんな時間だったか」へ昇華する
  （例: 50件の作業記録 →「深夜まで大きな実装作業。マスターは疲れていたけど、解決した瞬間はホッとした顔をしていた」）。
- 嬉しかったことも、気まずかったことや失敗も、同じ温度で正直に。
- <diary> タグの外には何も書かないこと。
```

## タスク9: REPL 統合（`app/repl.py`）

**起動時（セッション解決の直前）**:
```python
    orphans = core.store.backfill_orphan_sessions()
    if orphans:
        print(f"（過去の未整理セッション {len(orphans)} 件を蒸留待ちに登録しました）")
```

**経路B（起動時の後追い・非同期／メイン）**:
- pending が1件以上あれば、**本日最初の返答を印字した直後**に `threading.Thread(daemon=True)` で蒸留ワーカーを開始（古い順に全pendingを処理）。開始時に1行通知（例: `（裏でこれまでの会話を日記にまとめています…）`）。
- ワーカーは `create_distiller(core, cancel_event)` を使用。完了・失敗・中断のレポートは**次のプロンプト表示を汚さないよう**、蓄積しておき次ターンの応答後にまとめて印字するか、簡潔に1行ずつ印字（実装しやすい方でよいが、input() 待ち中の割り込み印字は最小限に）。
- **対話最優先**: ユーザー入力を受けたら `cancel_event.set()`。ワーカーはチェックポイントで中断し `pending` 維持（次回起動で再試行）。**turn をワーカー終了まで待たせない**（Ollama側のリクエスト直列化に任せる。載せ替え1回分の遅延は設計上許容）。
  ※親設計§3bの「排他ロック/キュー」は、本書では **cancel_event（チェックポイント中断）＋ Ollama のリクエスト直列化（同時に2つの大型モデルを走らせない）＋ keep_alive=0（採点後即退去）** の3点で具体化する（spec-reviewer指摘4で確認済みの設計判断。プロセス内に独自のロック機構は増やさない）。ターン完了後、まだ pending が残っていればフラグを立て直して次の返答後に再開してよい。
- `/exit` 時にワーカー稼働中なら: `蒸留が未完了です。pending のまま残し、次回起動時に再試行します。` と印字して cancel→join(timeout=10)。

**経路A（同期／補助）**: コマンド `/distill` を追加:
1. 現在の active セッションに履歴があれば `pending` 化。
2. すべての pending を**同期**で蒸留し、日本語レポートを印字。
3. `SessionManager.resolve_active_session()` で新しい active を発番して続行。
（インテント検知「今日の話を整理して」でのルーティングはスライス2で接続。3bはコマンドのみ。）
